Perbaikan PDF dengan Compact Header & Styling function generatePDF(type) { showToast(`Membuat laporan PDF ${type}...`); setTimeout(() => { const { jsPDF } = window.jspdf; const doc = new jsPDF('landscape'); doc.setFontSize(12); doc.text(`Laporan Jurnal Mengajar (${type.toUpperCase()})`, 14, 15); doc.setFontSize(10); doc.text(`Guru: ${profile.name} | NIP: ${profile.nip}`, 14, 22); const tableData = journalEntries.map(j => [ j.date, j.dayName, j.subject, j.topic, `${j.present}H / ${j.sick}S / ${j.permission}I / ${j.absent}A`, j.condition ]); doc.autoTable({ startY: 28, head: [['Tanggal', 'Hari', 'Mata Pelajaran', 'Topik Pembahasan', 'Kehadiran', 'Kondisi']], body: tableData, styles: { fontSize: 7.5, cellPadding: 2 }, headStyles: { fillColor: [23, 105, 209], textColor: 255, fontStyle: 'bold', halign: 'center' }, bodyStyles: { valign: 'middle' }, }); doc.save(`Jurnal_${profile.name}_${type}.pdf`); showToast("PDF berhasil diunduh!"); }, 1000); } // 2. Auto-Redirect setelah Simpan Jurnal function saveJournalEntry() { const subject = document.getElementById('journal-subject-select').value; const method = document.getElementById('journal-method-select').value; const condition = document.getElementById('journal-condition-select').value; const topic = document.getElementById('journal-topic').value; const notes = document.getElementById('journal-notes').value; const date = document.getElementById('journal-date-picker').value || new Date().toISOString().split('T')[0]; const entry = { id: currentEditingId || 'j_' + Date.now(), date, dayName: new Date(date).toLocaleDateString('id-ID', { weekday: 'long' }).toUpperCase(), subject, method, condition, topic, notes, present: parseInt(document.getElementById('att-present').value) || 0, sick: parseInt(document.getElementById('att-sick').value) || 0, permission: parseInt(document.getElementById('att-permission').value) || 0, absent: parseInt(document.getElementById('att-absent').value) || 0, photo: window.tempPhotoData || null }; if (currentEditingId) { const idx = journalEntries.findIndex(j => j.id === currentEditingId); if (idx !== -1) journalEntries[idx] = entry; } else { journalEntries.unshift(entry); } localStorage.setItem('jg_journals', JSON.stringify(journalEntries)); showToast("Jurnal berhasil disimpan!"); // Auto-Redirect ke Dashboard / Home Screen showScreen('home-screen'); renderHistory(); } // 3. Penghapusan Data Seketika (Real-time Delete) tanpa Refresh function deleteJournal(id) { if (!confirm("Hapus catatan jurnal ini?")) return; // Filter array langsung journalEntries = journalEntries.filter(j => j.id !== id); // Perbarui LocalStorage localStorage.setItem('jg_journals', JSON.stringify(journalEntries)); showToast("Jurnal berhasil dihapus."); // Render ulang komponen riwayat/tabel seketika renderHistory(); }
